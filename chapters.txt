def decode_binary(text: str) -> str:
    """
    Пытается декодировать бинарный код в текст.
    Возвращает декодированную строку или None если это не бинарный код.
    """
    # Убираем пробелы и проверяем что это только 0 и 1
    cleaned = text.replace(" ", "").replace("-", "").replace(".", "")
    
    if not cleaned:
        return None
    
    # Проверяем что строка состоит только из 0 и 1
    if not all(c in '01' for c in cleaned):
        return None
    
    # Должно быть кратно 8 (байты)
    if len(cleaned) % 8 != 0:
        # Пробуем дополнить нулями слева
        padding = 8 - (len(cleaned) % 8)
        cleaned = '0' * padding + cleaned
    
    if len(cleaned) < 8:
        return None
    
    try:
        # Разбиваем на байты и декодируем
        bytes_list = [cleaned[i:i+8] for i in range(0, len(cleaned), 8)]
        decoded = ''.join([chr(int(b, 2)) for b in bytes_list])
        
        # Проверяем что результат — читаемый текст
        if decoded.isprintable() or any(c.isalpha() for c in decoded):
            return decoded.lower()
        return None
    except:
        return None


def is_binary_name(name: str) -> bool:
    """Проверяет, является ли имя бинарным кодом"""
    cleaned = name.replace(" ", "").replace("-", "").replace(".", "")
    if len(cleaned) < 8:
        return False
    return all(c in '01' for c in cleaned) and len(cleaned) >= 8

    def step_7(self):
        """Создание защиты"""
        name = self.save.data["player_name"]

        shield_path = os.path.join(self.desktop.desktop_path, "ЗАЩИТА.txt")

        start = time.time()
        spam_count = 0
        found = False

        while time.time() - start < 180:
            # Проверяем файл
            if os.path.exists(shield_path):
                try:
                    with open(shield_path, 'r', encoding='utf-8') as f:
                        content = f.read().upper()
                    
                    if "ЗДЕСЬ" in content and "РЕАЛЬНА" in content and "ОШИБКА" in content:
                        print("\n[✓] Защита активирована!")
                        found = True
                        break
                except:
                    pass

            time.sleep(2)

            # Протокол создаёт мусорные файлы
            spam_count += 1
            if spam_count == 5:
                self.voice.protocol_window(
                    "ПРЕКРАТИ.\n\n"
                    "ЭТО БЕСПОЛЕЗНО.",
                    width=350, height=200
                )
            elif spam_count == 15:
                for i in range(3):
                    spam_name = f"ПРОТОКОЛ_{i+1}.txt"
                    self.desktop.create_file_on_desktop(
                        spam_name,
                        "ПРЕКРАТИ\nПРЕКРАТИ\nПРЕКРАТИ\n" * 20
                    )
                print("[!] Протокол создаёт мусорные файлы...")
                
                self.voice.protocol_window(
                    "Я ПРЕДУПРЕЖДАЛ.\n\n"
                    "ТЕПЕРЬ ПОСМОТРИМ\n"
                    "КТО СИЛЬНЕЕ.",
                    width=400, height=250
                )
            elif spam_count == 30:
                self.voice.say(
                    f"{name}!\n\nПоторопись!\nОн становится сильнее!",
                    title="Система!!"
                )

        if not found:
            # Создаём сами если не успел
            with open(shield_path, 'w', encoding='utf-8') as f:
                f.write("Я ЗДЕСЬ\nЯ РЕАЛЬНА\nЯ НЕ ОШИБКА")

                "Я создала защиту сама!\n\n"
                "Но она слабее\n"

                "Ладно. Попробуем.",

            )

            self.save.puzzle_solved()

    def show(self):

        screen_height = self.root.winfo_screenheight()

    def _show_sequence(self, sequence, index):
        if not self.running:
            return
        try:
            if not self.root.winfo_exists():
                return
        except:
            return
            
        if index >= len(sequence):
            self.root.after(2000, self._skip)
            return
        
        text, duration = sequence[index]
        self.label.config(text="")
        self._typewriter(text, 0, duration, sequence, index)
    
    def _typewriter(self, text, char_idx, duration, sequence, seq_idx):
        if not self.running:
            return
        try:
            if not self.root.winfo_exists():
                return
        except:
            return
            
        if char_idx <= len(text):
            self.label.config(text=text[:char_idx])
            delay = max(25, min(60, 800 // (len(text) + 1))) if text else 50
            self.root.after(delay, lambda: self._typewriter(text, char_idx + 1, duration, sequence, seq_idx))
        else:
            self.root.after(duration, lambda: self._show_sequence(sequence, seq_idx + 1))

def check_secret_ending(save) -> bool:
    """Секретная концовка: бинарное имя + 100% в каждой главе"""
    flags = save.data.get("flags", {})
    if not flags.get("binary_name", False):
        return False
    for i in range(5):
        if flags.get(f"chapter_{i}_relationship", 0) < 100:
            return False
    return True

def check_bad_ending(save) -> bool:
    """Плохая концовка: отношения < 20%"""
    return save.data.get("relationship", 50) < 20

def check_merge_ending(save) -> bool:
    """Бонусная концовка: 90%+ отношений, сторона Системы, 8+ головоломок"""
    rel = save.data.get("relationship", 50)
    flags = save.data.get("flags", {})
    puzzles = save.data.get("puzzles_solved", 0)
    return rel >= 90 and flags.get("sided_with") == "system" and puzzles >= 8

    def _create_ending_files(self, ending_name: str, description: str):
        """Создаёт файлы финала"""
        name = self.save.data["player_name"]
        rel = self.save.data.get("relationship", 50)
        puzzles = self.save.data.get("puzzles_solved", 0)

        folder = os.path.join(self.desktop.desktop_path, "ФИНАЛ")
        os.makedirs(folder, exist_ok=True)

        with open(os.path.join(folder, "ЭПИЛОГ.txt"), 'w', encoding='utf-8') as f:
            f.write(
                f"═══════════════════════════════════════\n"
                f"            {ending_name}\n"
                f"═══════════════════════════════════════\n\n"
                f"{description}\n\n"
                f"Игрок: {name}\n"
                f"Отношения: {rel}%\n"
                f"Головоломок: {puzzles}\n\n"
                f"Спасибо за игру.\n— Система\n"
            )

        self.desktop.create_file_on_desktop(
            f"ФИНАЛ_{ending_name.split()[0]}.txt",
            f"DESKVENTURE — ФИНАЛ\n{'='*40}\n\n"
            f"КОНЦОВКА: {ending_name}\n\n"
            f"Игрок: {name}\n"
            f"Отношения: {rel}%\n"
            f"Головоломок: {puzzles}\n\n"
            f"{description}\n"
        )

        # Обновляем достижения
        update_achievements(self.desktop, self.save, 5)
